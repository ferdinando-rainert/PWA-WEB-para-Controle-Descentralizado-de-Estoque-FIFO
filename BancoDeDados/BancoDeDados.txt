-- =====================================================================
--  BANCO DE DADOS: ESTOQUE FIFO (MySQL 8.0.16 ou superior)
--  Controle descentralizado de estoque com regra FIFO
-- =====================================================================

CREATE DATABASE IF NOT EXISTS estoque_fifo
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_0900_ai_ci;

USE estoque_fifo;

-- ---------------------------------------------------------------------
-- PRODUTO
-- ---------------------------------------------------------------------
CREATE TABLE produto (
  id INT PRIMARY KEY AUTO_INCREMENT,
  sku VARCHAR(40) NOT NULL UNIQUE,
  nome VARCHAR(120) NOT NULL,
  unidade_medida VARCHAR(10) NOT NULL,
  estoque_minimo DECIMAL(14,3) NOT NULL DEFAULT 0,
  ativo BOOLEAN NOT NULL DEFAULT TRUE
);

-- ---------------------------------------------------------------------
-- LOTE (cada entrada de produto gera um lote; data_entrada define o FIFO)
-- ---------------------------------------------------------------------
CREATE TABLE lote (
  id INT PRIMARY KEY AUTO_INCREMENT,
  produto_id INT NOT NULL,
  codigo_lote VARCHAR(40) NOT NULL,
  data_entrada DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  data_validade DATE,
  quantidade_inicial DECIMAL(14,3) NOT NULL,
  FOREIGN KEY (produto_id) REFERENCES produto(id),
  INDEX idx_lote_fifo (produto_id, data_entrada)
);

-- ---------------------------------------------------------------------
-- LOCAL_ESTOQUE (unidade ou setor: controle descentralizado)
-- ---------------------------------------------------------------------
CREATE TABLE local_estoque (
  id INT PRIMARY KEY AUTO_INCREMENT,
  nome VARCHAR(80) NOT NULL UNIQUE,
  descricao VARCHAR(255),
  ativo BOOLEAN NOT NULL DEFAULT TRUE
);

-- ---------------------------------------------------------------------
-- USUARIO
-- ---------------------------------------------------------------------
CREATE TABLE usuario (
  id INT PRIMARY KEY AUTO_INCREMENT,
  nome VARCHAR(100) NOT NULL,
  email VARCHAR(120) NOT NULL UNIQUE,
  senha_hash VARCHAR(255) NOT NULL,
  perfil ENUM('ADMIN','GESTOR','OPERADOR') NOT NULL DEFAULT 'OPERADOR',
  ativo BOOLEAN NOT NULL DEFAULT TRUE
);

-- ---------------------------------------------------------------------
-- SALDO_LOTE_LOCAL (quanto de cada lote existe em cada local)
-- ---------------------------------------------------------------------
CREATE TABLE saldo_lote_local (
  lote_id INT NOT NULL,
  local_id INT NOT NULL,
  quantidade DECIMAL(14,3) NOT NULL DEFAULT 0,
  PRIMARY KEY (lote_id, local_id),
  FOREIGN KEY (lote_id) REFERENCES lote(id),
  FOREIGN KEY (local_id) REFERENCES local_estoque(id),
  CONSTRAINT chk_saldo_nao_negativo CHECK (quantidade >= 0)
);

-- ---------------------------------------------------------------------
-- MOVIMENTACAO (histórico: nunca é apagado em operações comuns)
-- uuid_cliente evita duplicidade quando o PWA sincroniza dados offline
-- fifo_ignorado + justificativa registram saídas fora da ordem FIFO
-- ---------------------------------------------------------------------
CREATE TABLE movimentacao (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  uuid_cliente CHAR(36) NOT NULL UNIQUE,
  tipo ENUM('ENTRADA','SAIDA','TRANSFERENCIA','AJUSTE') NOT NULL,
  lote_id INT NOT NULL,
  local_origem_id INT NULL,
  local_destino_id INT NULL,
  quantidade DECIMAL(14,3) NOT NULL,
  usuario_id INT NOT NULL,
  data_hora_cliente DATETIME NOT NULL,
  fifo_ignorado BOOLEAN NOT NULL DEFAULT FALSE,
  justificativa VARCHAR(255),
  FOREIGN KEY (lote_id) REFERENCES lote(id),
  FOREIGN KEY (local_origem_id) REFERENCES local_estoque(id),
  FOREIGN KEY (local_destino_id) REFERENCES local_estoque(id),
  FOREIGN KEY (usuario_id) REFERENCES usuario(id),
  CONSTRAINT chk_mov_quantidade_positiva CHECK (quantidade > 0)
);
